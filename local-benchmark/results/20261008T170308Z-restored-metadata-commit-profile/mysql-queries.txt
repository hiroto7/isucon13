SELECT u.name AS username,
		IFNULL(r.reactions, 0) + IFNULL(c.tips, 0) AS score
		FROM users u
		LEFT JOIN (SELECT l.user_id, COUNT(*) AS reactions
			FROM livestreams l INNER JOIN reactions r ON r.livestream_id = l.id
			GROUP BY l.user_id) r ON r.user_id = u.id
		LEFT JOIN (SELECT l.user_id, SUM(c.tip) AS tips
			FROM livestreams l INNER JOIN livecomments c ON c.livestream_id = l.id
			GROUP BY l.user_id) c ON c.user_id = u.id\G
